{{- /* Theme override. The trail is drawn as a filesystem path — every page rendered as the .md file it lives in — instead of a chevron list of titles. A leaf page is .md; a section is its own folder PLUS a self-named index file, so /learn/linux/ reads /home/csc/learn/linux/linux.md and its child reads /home/csc/learn/linux/filesystem.md. Ancestor sections stay folders, and every segment links to its own directory the way a file manager's path bar works. The current page's file segment links to itself and is marked aria-current. Folder/file names are URL slugs, not titles, because a directory is named by its folder and "Permissions & Users" is not a folder name. The site root is the one exception: it prints /home/csc, or the club's full name on the home page itself, where the name is the subject rather than a location. Hrefs are .RelPermalink (clean "/learn/linux/" directory URLs) rather than the theme's permalink helper, which appends index.html. Emits schema.org BreadcrumbList microdata over the real trail (the extra index-file crumb a section adds is visual only), and still honours dirOnly (the children shortcode plainifies this into a parent label). */ -}} {{- $page := .page }} {{- if .dirOnly }}{{ $page = $page.Parent }}{{ end }} {{- with $page }} {{- if not (partial "_relearn/pageIsSpecial.gotmpl" .) }} {{- $trail := .Ancestors.Reverse | append . }} {{- $last := sub (len $trail) 1 }} {{- /* A current section emits two crumbs (its folder, then its .md index file), so build the list first. schema ListItems cover the real trail only; the extra file crumb is schema=false with no position. */ -}} {{- $crumbs := slice }} {{- range $i, $p := $trail }} {{- $isLast := eq $i $last }} {{- $url := $p.RelPermalink }} {{- $slug := path.Base $p.RelPermalink }} {{- $pos := add $i 1 }} {{- if $p.IsHome }} {{- $crumbs = $crumbs | append (dict "url" $url "label" (cond $isLast site.Title "/home/csc") "current" $isLast "schema" $.schema "pos" $pos "slash" (cond $isLast "" "/")) }} {{- else if not $isLast }} {{- $crumbs = $crumbs | append (dict "url" $url "label" $slug "current" false "schema" $.schema "pos" $pos "slash" "/") }} {{- else if $p.IsSection }} {{- $crumbs = $crumbs | append (dict "url" $url "label" $slug "current" false "schema" $.schema "pos" $pos "slash" "/") }} {{- $crumbs = $crumbs | append (dict "url" $url "label" (printf "%s.md" $slug) "current" true "schema" false "pos" 0 "slash" "") }} {{- else }} {{- $crumbs = $crumbs | append (dict "url" $url "label" (printf "%s.md" $slug) "current" true "schema" $.schema "pos" $pos "slash" "") }} {{- end }} {{- end }} {{- /* One line, no whitespace between
  • : .breadcrumbs li is inline-block, so a newline between two would render as a space and split the path. */ -}} {{- range $c := $crumbs }}{{ $c.label }}{{ if $c.schema }}{{ end }}{{ $c.slash }}
