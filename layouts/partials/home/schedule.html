{{- /* Semester schedule. Source of truth is the home page's own front matter (content/_index.md → [params.schedule] and [[params.sessions]]), so the club edits the calendar as Markdown and the page applies it. Row state is derived from the date. `status` in front matter only overrides. Dates are "YYYY-MM-DD" strings, so lexicographic compare is chronological. */ -}} {{- $sched := .Params.schedule | default dict }} {{- /* Sessions inherit the weekly room unless they override it. Most weeks are in the usual place; the ones that are not are why this column exists — a different building, or "Online". */ -}} {{- $defaultRoom := (.Params.meeting | default dict).room | default "" }} {{- $sessions := sort (.Params.sessions | default slice) "date" }} {{- with $sessions }} {{- $today := now.Format "2006-01-02" }} {{- /* Is one of today's dates a live meeting? If so it takes the highlight and nothing gets flagged NEXT. */ -}} {{- $hasToday := false }} {{- range $sessions }} {{- $skip := in (slice "cancelled" "moved") (.status | default "") }} {{- if and (not $skip) (eq .date $today) }}{{ $hasToday = true }}{{ end }} {{- end }} {{- /* NEXT = earliest session still ahead of today that has not been called off. Only assigned when today is not itself a meeting day. */ -}} {{- /* Track the NEXT row by INDEX, not by date: two meetings can share a date, and matching on the date value flags both of them. */ -}} {{- $nextIdx := -1 }} {{- if not $hasToday }} {{- range $i, $sn := $sessions }} {{- $skip := in (slice "cancelled" "moved") ($sn.status | default "") }} {{- if and (eq $nextIdx -1) (not $skip) (gt $sn.date $today) }} {{- $nextIdx = $i }} {{- end }} {{- end }} {{- end }}
